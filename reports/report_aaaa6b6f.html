📊 Approach & Query

Query Code


-- Get all normalized categories ranked for the summary table
WITH normalized_categories AS (
  SELECT 
    region,
    CASE 
      WHEN LOWER(category) LIKE '%electronic%' THEN 'Electronics'
      WHEN LOWER(category) LIKE '%apparel%' OR LOWER(category) LIKE '%clothing%' THEN 'Apparel & Clothing'
      WHEN LOWER(category) LIKE '%beauty%' OR LOWER(category) LIKE '%cosmetic%' THEN 'Beauty & Cosmetics'
      WHEN LOWER(category) LIKE '%sport%' THEN 'Sports & Outdoor'
      WHEN LOWER(category) LIKE '%home%' OR LOWER(category) LIKE '%kitchen%' THEN 'Home & Kitchen'
      ELSE category
    END as normalized_category,
    COUNT(*) as transaction_count
  FROM gold_sales_by_category_region
  WHERE region IS NOT NULL AND category IS NOT NULL
  GROUP BY region, CASE 
      WHEN LOWER(category) LIKE '%electronic%' THEN 'Electronics'
      WHEN LOWER(category) LIKE '%apparel%' OR LOWER(category) LIKE '%clothing%' THEN 'Apparel & Clothing'
      WHEN LOWER(category) LIKE '%beauty%' OR LOWER(category) LIKE '%cosmetic%' THEN 'Beauty & Cosmetics'
      WHEN LOWER(category) LIKE '%sport%' THEN 'Sports & Outdoor'
      WHEN LOWER(category) LIKE '%home%' OR LOWER(category) LIKE '%kitchen%' THEN 'Home & Kitchen'
      ELSE category
    END
),
ranked_categories AS (
  SELECT 
    region,
    normalized_category as category,
    transaction_count,
    RANK() OVER (PARTITION BY region ORDER BY transaction_count DESC) as revenue_rank,
    ROUND(100.0 * transaction_count / SUM(transaction_count) OVER (PARTITION BY region), 2) as pct_of_region
  FROM normalized_categories
)
SELECT 
  region,
  category,
  transaction_count,
  revenue_rank,
  pct_of_region
FROM ranked_categories
ORDER BY region, revenue_rank

Query Description

SQL query normalized category names (consolidating variations like 'Electronics', 'ELECTRONICS', 'Elec.' into single categories), aggregated transaction counts by region and category, applied RANK() window function to create region-specific rankings (PARTITION BY region ORDER BY transaction_count DESC), and calculated percentage share of total regional transactions (pct_of_region).
